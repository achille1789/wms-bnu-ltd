+    /**

+        this.name = name;

+        this.quantity = quantity;
+        this.supplier = supplier;

+        goodData.put(Data.SUPPLIER, this.supplier);
+        return goodData;
+    }

+        this.quantity = quantity;
+    }

+     * Getters
+     */

+        return this.name;
+    }
+
